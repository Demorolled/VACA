## Code Chunk (single responsibility)

```ts
export function vocabList(words: string[]) {
  const unique = [...new Set(words.map((w) => w.trim()).filter(Boolean))].sort((a, b) => a.localeCompare(b));
  return { words: unique, count: unique.length };
}
```
