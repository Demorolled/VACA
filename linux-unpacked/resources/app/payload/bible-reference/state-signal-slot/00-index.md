```tsx
export function createSignal<T = void>() {
  const slots = new Set<(payload: T) => void>();

  return {
    connect(fn) {
      slots.add(fn);
      return () => slots.delete(fn);
    },
    emit(payload) {
      for (const fn of [...slots]) fn(payload);
    },
    get count() { return slots.size; },
    disconnectAll() { slots.clear(); },
  };
}
```
