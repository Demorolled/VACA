export function maskEmail(email: string) {
  const at = email.indexOf('@');
  if (at <= 0) return '***';
  const local = email.slice(0, at);
  const domain = email.slice(at + 1);
  const shown = local.charAt(0) + '***';
  return `${shown}@${domain}`;
}
