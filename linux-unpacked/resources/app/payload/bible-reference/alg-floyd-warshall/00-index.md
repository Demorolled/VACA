```tsx
export function floydWarshall(adj: number[][]) {
  const n = adj.length;
  const dist = adj.map((row) => row.slice());
  for (let k = 0; k < n; k++)
    for (let i = 0; i < n; i++)
      for (let j = 0; j < n; j++)
        if (dist[i][k] + dist[k][j] < dist[i][j]) dist[i][j] = dist[i][k] + dist[k][j];
  return dist;
}
```
