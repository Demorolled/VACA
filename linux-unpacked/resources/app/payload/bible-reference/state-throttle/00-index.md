```tsx
export function throttle<A extends unknown[]>(fn: (...args: A) => void, intervalMs: number) {
  let last = 0;
  let timer: ReturnType<typeof setTimeout> | undefined;

  return (...args: A) => {
    const now = Date.now();
    const remaining = intervalMs - (now - last);
    if (remaining <= 0) {
      last = now;
      fn(...args);
    } else if (!timer) {
      timer = setTimeout(() => {
        timer = undefined;
        last = Date.now();
        fn(...args);
      }, remaining);
    }
  };
}
```
