## Code Chunk (single responsibility)

```ts
export function pwmTiming(periodUs: number, dutyPercent: number) {
  const d = Math.max(0, Math.min(100, dutyPercent)) / 100;
  const onUs = periodUs * d;
  return { onUs, offUs: periodUs - onUs };
}
```
