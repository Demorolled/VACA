```tsx
export function resolvePath(path: string) {
  const isAbs = path.startsWith('/');
  const parts = path.split('/').filter((p) => p && p !== '.');
  const out: string[] = [];
  for (const p of parts) {
    if (p === '..') out.pop();
    else out.push(p);
  }
  return (isAbs ? '/' : '') + out.join('/');
}
```
