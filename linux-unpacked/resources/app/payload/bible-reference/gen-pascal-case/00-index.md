export function toPascalCase(input: string) {
  const camel = input.split(/[^a-z0-9]+/i).filter(Boolean)
    .map((p) => p[0].toUpperCase() + p.slice(1).toLowerCase()).join('');
  return camel ? camel[0].toUpperCase() + camel.slice(1) : camel;
}
