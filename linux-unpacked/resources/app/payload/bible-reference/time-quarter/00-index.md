## Code Chunk (single responsibility)

```ts
export function quarterOf(date: Date, fiscalOffsetMonths = 0) {
  const d = new Date(date);
  d.setMonth(d.getMonth() - fiscalOffsetMonths);
  const quarter = Math.floor(d.getMonth() / 3) + 1;
  return { quarter, year: d.getFullYear(), label: `Q${quarter} ${d.getFullYear()}` };
}
```
