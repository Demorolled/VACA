## Code Chunk (single responsibility)

```tsx
export async function runMigrations(migrations: Migration[], applied: Set<number>) {
  const pending = migrations
    .filter((m) => !applied.has(m.version))
    .sort((a, b) => a.version - b.version);
  for (const m of pending) await m.up();
  return pending;
}
```
