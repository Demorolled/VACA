```tsx
export function safeFilename(name: string, maxLength = 100) {
  const cleaned = name
    .replace(/[\/\\:*?"<>|\x00-\x1f]/g, '_')
    .replace(/^\.+/, '')
    .replace(/[. ]+$/g, '')
    .slice(0, maxLength)
    .trim();
  return cleaned || 'untitled';
}
```
