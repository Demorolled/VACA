## Code Chunk (single responsibility)

```tsx
const CODES: Record<string, number> = { red: 31, green: 32, yellow: 33, blue: 34, magenta: 35, cyan: 36, white: 37, gray: 90, bold: 1, dim: 2 };
export class Ansi {
  constructor(private enabled = true) {}
  paint(color: string, text: string) {
    const code = CODES[color];
    if (!this.enabled || code === undefined) return text;
    return `\u001b[${code}m${text}\u001b[0m`;
  }
}
```
