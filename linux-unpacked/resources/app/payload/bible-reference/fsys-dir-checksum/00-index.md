## Code Chunk (single responsibility)

```tsx
export function dirFingerprint(files: Array<{ path: string; hash: string }>, hashFn: (s: string) => string) {
  const lines = files.map((f) => `${f.path}:${f.hash}`).sort().join('\n');
  return hashFn(lines);
}
```
