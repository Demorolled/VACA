export function parseTscErrors(output: string) {
  const out: TsError[] = [];
  const re = /^(.+?)\((\d+),(\d+)\):\s+error\s+(TS\d+):\s+(.*)$/gm;
  let m: RegExpExecArray | null;
  while ((m = re.exec(output))) out.push({ file: m[1], line: Number(m[2]), col: Number(m[3]), code: m[4], message: m[5] });
  return out;
}
