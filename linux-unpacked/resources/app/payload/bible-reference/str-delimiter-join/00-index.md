## Code Chunk (single responsibility)

```ts
export function joinNonEmpty(parts: string[], delimiter = ', ') {
  return parts.filter((p) => p !== undefined && p !== null && p !== '').join(delimiter);
}
export function splitKeepTrimmed(text: string, delimiter: string) {
  return text.split(delimiter).map((p) => p.trim()).filter(Boolean);
}
```
