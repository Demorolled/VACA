```tsx
export function humanBytes(bytes: number, decimals = 1) {
  if (!Number.isFinite(bytes)) return '0 B';
  const units = ['B', 'KB', 'MB', 'GB', 'TB', 'PB'];
  let u = 0, v = bytes;
  while (Math.abs(v) >= 1000 && u < units.length - 1) { v /= 1000; u++; }
  return `${v.toFixed(u === 0 ? 0 : decimals)} ${units[u]}`;
}
```
