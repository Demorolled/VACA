export function parseConfirm(input: string | undefined, def = false) {
  if (!input) return def;
  const v = input.trim().toLowerCase();
  if (['y', 'yes', 'true', '1'].includes(v)) return true;
  if (['n', 'no', 'false', '0'].includes(v)) return false;
  return def;
}
