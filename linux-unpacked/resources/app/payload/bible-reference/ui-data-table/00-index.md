**Why:** Atomic table: columns + rows in, onSort out; the table owns only layout, never data fetching

## Standardized Interface (contract)

export interface TableColumn<K extends string> { key: K; header: string; width?: number }
export interface DataTableProps<T, K extends string = Extract<keyof T, string>> {
  columns: TableColumn<K>[];
  rows: T[];
  sortKey?: K;
  sortDir?: 'asc' | 'desc';
  onSort?: (key: K) => void;
}

## Code Chunk (single responsibility)

```tsx
export function DataTable<T extends Record<string, unknown>, K extends string = Extract<keyof T, string>>(
  props: DataTableProps<T, K>,
) {
  return (
    <table className="data-table">
      <thead>
        <tr>
          {props.columns.map((c) => (
            <th key={c.key} style={c.width ? { width: c.width } : undefined}>
              {props.onSort ? (
                <button type="button" className="th-sort" onClick={() => props.onSort!(c.key)}>
                  {c.header}
                  {props.sortKey === c.key ? (props.sortDir === 'asc' ? ' ▲' : ' ▼') : ''}
                </button>
              ) : (
                c.header
              )}
            </th>
          ))}
        </tr>
      </thead>
      <tbody>
        {props.rows.map((row, i) => (
          <tr key={i}>
            {props.columns.map((c) => (
              <td key={c.key}>{String(row[c.key] ?? '')}</td>
            ))}
          </tr>
        ))}
      </tbody>
    </table>
  );
}
```
