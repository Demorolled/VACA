export function menuLines(options: string[]) {
  return options.map((o, i) => `  ${i + 1}. ${o}`);
}
export function menuSelect(input: string, count: number) {
  const n = parseInt(input, 10);
  return Number.isFinite(n) && n >= 1 && n <= count ? n - 1 : null;
}
