## Code Chunk (single responsibility)

```tsx
const IRREGULAR: Record<string, string> = { person: 'people', child: 'children', mouse: 'mice', foot: 'feet', tooth: 'teeth', goose: 'geese', man: 'men', woman: 'women' };
export function pluralize(word: string) {
  if (IRREGULAR[word.toLowerCase()]) return IRREGULAR[word.toLowerCase()];
  if (/(s|x|z|ch|sh)$/i.test(word)) return word + 'es';
  if (/[^aeiou]y$/i.test(word)) return word.slice(0, -1) + 'ies';
  if (/(f|fe)$/i.test(word)) return word.replace(/f{1,2}e?$/i, 'ves');
  return word + 's';
}
```
