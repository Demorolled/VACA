```tsx
export function expandPath(path: string, env: Record<string, string> = (process?.env ?? {}) as Record<string, string>) {
  let out = path;
  if (out.startsWith('~')) out = (env.HOME ?? env.USERPROFILE ?? '') + out.slice(1);
  out = out.replace(/\$(\w+)/g, (_, name) => env[name] ?? '');
  return out;
}
```
