```tsx
export function formatCompact(value: number, decimals = 1) {
  const abs = Math.abs(value);
  const units = ['', 'K', 'M', 'B', 'T'];
  let u = 0;
  while (abs >= 1000 && u < units.length - 1) { value /= 1000; u++; }
  return `${value.toFixed(value >= 100 || u === 0 ? 0 : decimals)}${units[u]}`;
}
```
