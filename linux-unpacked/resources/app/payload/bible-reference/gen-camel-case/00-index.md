export function toCamelCase(input: string) {
  const parts = input.split(/[^a-z0-9]+/i).filter(Boolean);
  return parts.map((p, i) => (i === 0 ? p.toLowerCase() : p[0].toUpperCase() + p.slice(1).toLowerCase())).join('');
}
