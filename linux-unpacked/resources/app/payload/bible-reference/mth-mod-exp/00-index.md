```ts
export function modExp(base: number, exp: number, mod: number) {
  if (mod === 0) return NaN;
  let result = 1n, b = BigInt(base) % BigInt(mod), e = BigInt(exp), m = BigInt(mod);
  while (e > 0n) { if (e & 1n) result = (result * b) % m; b = (b * b) % m; e >>= 1n; }
  return Number(result);
}
```
