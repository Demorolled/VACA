export function escapeSqlLiteral(value: string) {
  return value.replace(/[\x00\x08\x09\x1a\n\r"'\\%_]/g, (ch) => {
    switch (ch) {
      case "'": return "''";
      case '\\': return '\\\\';
      case '\n': return '\\n';
      case '\r': return '\\r';
      default: return ch;
    }
  });
}
