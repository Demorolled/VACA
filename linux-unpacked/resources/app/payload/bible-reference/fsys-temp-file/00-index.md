```tsx
export function tempName(prefix = 'tmp', extension = 'tmp', existing: Set<string> = new Set()) {
  let name = '';
  do {
    name = `${prefix}-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}.${extension}`;
  } while (existing.has(name));
  existing.add(name);
  return name;
}
```
