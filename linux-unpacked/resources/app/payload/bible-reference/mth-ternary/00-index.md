export function ternaryMax(f: (x: number) => number, lo: number, hi: number, tol = 1e-9) {
  let l = lo, h = hi;
  while (h - l > tol) {
    const m1 = l + (h - l) / 3, m2 = h - (h - l) / 3;
    if (f(m1) < f(m2)) l = m1; else h = m2;
  }
  const x = (l + h) / 2;
  return { x, value: f(x) };
}
