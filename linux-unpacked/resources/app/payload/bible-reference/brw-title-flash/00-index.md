## Code Chunk (single responsibility)

```tsx
export function flashTitle(message: string, originalTitle = document.title) {
  let shown = false;
  const interval = setInterval(() => {
    document.title = shown ? originalTitle : message;
    shown = !shown;
  }, 1000);
  return () => { clearInterval(interval); document.title = originalTitle; };
}
```
