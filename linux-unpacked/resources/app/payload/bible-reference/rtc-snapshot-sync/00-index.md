```tsx
export function mergeSync<T>(local: SyncState<T>, incoming: SyncState<T> & { deltas?: Array<{ version: number; apply: (d: T) => T }> }) {
  let data = local.data;
  let applied = 0;
  if (incoming.version > local.version) {
    if (incoming.deltas) {
      for (const d of incoming.deltas) { if (d.version > local.version) { data = d.apply(data); applied++; } }
    } else {
      data = incoming.data;
      applied = 1;
    }
    return { state: { version: Math.max(local.version, incoming.version), data }, applied };
  }
  return { state: local, applied: 0 };
}
```
