export function stackSeries(series: number[][]) {
  const n = series[0]?.length ?? 0;
  const acc = new Array(n).fill(0);
  return series.map((s) => s.map((v, i) => {
    const y0 = acc[i];
    acc[i] += v;
    return { y0, y1: acc[i] };
  }));
}
