```tsx
export function binarySearch<T>(sorted: T[], target: T, compare?: (a: T, b: T) => number): number {
  let lo = 0, hi = sorted.length - 1;
  const cmp = compare ?? ((a: T, b: T) => (a < b ? -1 : a > b ? 1 : 0));
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    const c = cmp(sorted[mid], target);
    if (c === 0) return mid;
    if (c < 0) lo = mid + 1;
    else hi = mid - 1;
  }
  return -1;
}

export function lowerBound<T>(sorted: T[], target: T, compare?: (a: T, b: T) => number): number {
  let lo = 0, hi = sorted.length;
  const cmp = compare ?? ((a: T, b: T) => (a < b ? -1 : a > b ? 1 : 0));
  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    if (cmp(sorted[mid], target) < 0) lo = mid + 1;
    else hi = mid;
  }
  return lo;
}
```
