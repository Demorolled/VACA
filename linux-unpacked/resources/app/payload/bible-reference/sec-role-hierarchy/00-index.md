```tsx
export function expandRoles(role: string, parents: Record<string, string[]>) {
  const seen = new Set<string>();
  const visit = (r: string) => {
    if (seen.has(r)) return;
    seen.add(r);
    for (const p of parents[r] ?? []) visit(p);
  };
  visit(role);
  return [...seen];
}
```
