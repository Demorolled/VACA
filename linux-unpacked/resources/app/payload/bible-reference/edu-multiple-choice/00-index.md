```ts
export function scoreMultipleChoice(answers: string[], key: string[], partialCredit = false) {
  let score = 0;
  const wrong: number[] = [];
  answers.forEach((a, i) => {
    if (a === key[i]) score++;
    else if (partialCredit && a) score += 0.5;
    else wrong.push(i);
  });
  const total = key.length;
  return { correct: Math.round(score), total, percent: total === 0 ? 0 : (score / total) * 100, wrong };
}
```
