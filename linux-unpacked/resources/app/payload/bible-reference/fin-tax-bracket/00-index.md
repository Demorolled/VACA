```tsx
export function taxBracketTax(income: number, brackets: TaxBracket[]) {
  let tax = 0, prev = 0;
  for (const b of brackets) {
    const slice = Math.min(income, b.upTo) - prev;
    if (slice <= 0) break;
    tax += slice * b.rate;
    prev = b.upTo;
  }
  return tax;
}
```
