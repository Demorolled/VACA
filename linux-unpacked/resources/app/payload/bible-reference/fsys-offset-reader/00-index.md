## Code Chunk (single responsibility)

```tsx
export async function readRange(read: (offset: number, length: number) => Promise<Uint8Array>, total: number, offset: number, length: number, decode: (b: Uint8Array) => string = (b) => new TextDecoder().decode(b)) {
  const start = Math.max(0, Math.min(offset, total));
  const end = Math.min(total, start + Math.max(0, length));
  if (start >= end) return '';
  return decode(await read(start, end - start));
}
```
