## Code Chunk (single responsibility)

```tsx
export function filterList<T>(items: T[], query: string, label: (t: T) => string) {
  const q = query.trim().toLowerCase();
  if (!q) return items;
  return items.filter((t) => label(t).toLowerCase().includes(q));
}
```
