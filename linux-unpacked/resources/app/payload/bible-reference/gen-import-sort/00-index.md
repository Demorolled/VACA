```ts
export function sortImports(imports: string[]) {
  return imports.slice().sort((a, b) => {
    const pa = /from\s+['"]([^'"]+)['"]/.exec(a)?.[1] ?? a;
    const pb = /from\s+['"]([^'"]+)['"]/.exec(b)?.[1] ?? b;
    return pa.localeCompare(pb);
  });
}
```
