export interface BootCheck { name: string; run: () => boolean }
export function bootSelfTest(checks: BootCheck[]): BootResult {
  const results = checks.map((c) => ({ name: c.name, ok: c.run() }));
  return { checks: results, ok: results.every((r) => r.ok) };
}
