## Code Chunk (single responsibility)

```tsx
export function watchElementSize(el: Element, onResize: (size: { width: number; height: number }) => void) {
  if (typeof ResizeObserver === 'undefined') return () => {};
  const ro = new ResizeObserver((entries) => {
    for (const e of entries) onResize({ width: e.contentRect.width, height: e.contentRect.height });
  });
  ro.observe(el);
  return () => ro.disconnect();
}
```
