## Code Chunk (single responsibility)

```tsx
export function debtSnowball(debts: Array<{ name: string; balance: number; rate: number }>) {
  return [...debts]
    .sort((a, b) => a.balance - b.balance)
    .map((d, i) => ({ ...d, payoffOrder: i + 1 }));
}
```
