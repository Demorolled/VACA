export function shouldRotate(opts: { size: number; maxBytes: number; ageMs: number; maxAgeMs: number; rotatedAt?: number }) {
  if (opts.size >= opts.maxBytes) return true;
  const age = Date.now() - (opts.rotatedAt ?? Date.now());
  return age >= opts.maxAgeMs;
}
