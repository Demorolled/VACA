## Code Chunk (single responsibility)

export class Flyweight<K, V> {
  private map = new Map<K, V>();
  get(key: K, create: () => V) {
    let v = this.map.get(key);
    if (v === undefined) { v = create(); this.map.set(key, v); }
    return v;
  }
  get size() { return this.map.size; }
}
