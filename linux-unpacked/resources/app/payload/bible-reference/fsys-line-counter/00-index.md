## Code Chunk (single responsibility)

```tsx
export function countLines(chunks: Iterable<string>) {
  let lines = 0, chars = 0;
  for (const c of chunks) {
    chars += c.length;
    lines += (c.match(/\n/g) ?? []).length;
  }
  return { lines, chars };
}
```
