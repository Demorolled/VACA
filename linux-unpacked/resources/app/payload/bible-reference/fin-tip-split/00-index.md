export function billSplit(amount: number, tipPercent: number, people: number, taxPercent = 0) {
  const tax = amount * (taxPercent / 100);
  const tip = (amount + tax) * (tipPercent / 100);
  const total = amount + tax + tip;
  return { subtotal: amount, tip, total, perPerson: people <= 0 ? total : total / people };
}
