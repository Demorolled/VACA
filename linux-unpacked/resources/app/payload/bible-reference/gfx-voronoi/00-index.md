```tsx
export function voronoiAssign(seeds: Array<{ x: number; y: number }>, points: Array<{ x: number; y: number }>) {
  return points.map((p) => {
    let best = 0, bd = Infinity;
    seeds.forEach((s, i) => {
      const d = (p.x - s.x) ** 2 + (p.y - s.y) ** 2;
      if (d < bd) { bd = d; best = i; }
    });
    return best;
  });
}
```
