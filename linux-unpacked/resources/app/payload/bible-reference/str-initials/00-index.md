## Code Chunk (single responsibility)

```ts
export function initials(name: string, max = 2) {
  return name
    .split(/\s+/)
    .filter(Boolean)
    .map((w) => w[0])
    .slice(0, max)
    .join('')
    .toUpperCase();
}
```
