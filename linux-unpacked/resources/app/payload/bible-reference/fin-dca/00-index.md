## Code Chunk (single responsibility)

```tsx
export function dollarCostAverage(periodicAmount: number, prices: number[]) {
  let shares = 0, spent = 0;
  for (const price of prices) {
    if (price <= 0) continue;
    shares += periodicAmount / price;
    spent += periodicAmount;
  }
  return { shares, totalSpent: spent, avgCost: spent / (shares || 1) };
}
```
