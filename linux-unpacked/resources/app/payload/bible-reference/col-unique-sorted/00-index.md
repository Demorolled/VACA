```tsx
export function sortedInsert<T>(arr: T[], value: T, compare: (a: T, b: T) => number = (a, b) => (a < b ? -1 : a > b ? 1 : 0)) {
  let lo = 0, hi = arr.length;
  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    const c = compare(value, arr[mid]);
    if (c === 0) return false;
    if (c < 0) hi = mid; else lo = mid + 1;
  }
  arr.splice(lo, 0, value);
  return true;
}
```
