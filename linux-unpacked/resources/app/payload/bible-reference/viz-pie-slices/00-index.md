```ts
export function pieSlices(values: number[]) {
  const total = values.reduce((s, v) => s + Math.max(0, v), 0) || 1;
  const out: Slice[] = [];
  let a = -Math.PI / 2;
  for (const v of values) {
    const sweep = (Math.max(0, v) / total) * Math.PI * 2;
    out.push({ start: a, end: a + sweep, mid: a + sweep / 2 });
    a += sweep;
  }
  return out;
}
```
