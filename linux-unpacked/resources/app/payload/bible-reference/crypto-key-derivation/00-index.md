import * as crypto from 'crypto';

export function deriveKey(password: string, salt: string, options?: { iterations?: number; keyLen?: number }): Buffer {
  return crypto.pbkdf2Sync(
    password,
    salt,
    options?.iterations ?? 210_000,
    options?.keyLen ?? 32,
    'sha256',

}
