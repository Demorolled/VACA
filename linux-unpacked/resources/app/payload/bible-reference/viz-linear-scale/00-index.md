```ts
export function linearScale(domain: [number, number], range: [number, number], opts?: { clamp?: boolean; nice?: boolean }) {
  let [d0, d1] = domain, [r0, r1] = range;
  const f = ((v: number) => {
    let t = d1 === d0 ? 0.5 : (v - d0) / (d1 - d0);
    if (opts?.clamp) t = Math.max(0, Math.min(1, t));
    return r0 + t * (r1 - r0);
  }) as LinearScale;
  f.domain = [d0, d1]; f.range = [r0, r1];
  return f;
}
```
