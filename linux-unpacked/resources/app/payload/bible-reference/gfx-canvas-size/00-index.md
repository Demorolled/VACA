```ts
export function setupHiDpi(canvas: HTMLCanvasElement, w: number, h: number, dpr = window.devicePixelRatio || 1) {
  canvas.width = w * dpr;
  canvas.height = h * dpr;
  canvas.style.width = `${w}px`;
  canvas.style.height = `${h}px`;
  const ctx = canvas.getContext('2d')!;
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  return ctx;
}
```
