```tsx
export function interleave(...strings: string[]) {
  const cols = strings.map((s) => Array.from(s));
  const max = Math.max(...cols.map((c) => c.length), 0);
  const out: string[] = [];
  for (let i = 0; i < max; i++) for (const col of cols) if (i < col.length) out.push(col[i]);
  return out.join('');
}
```
