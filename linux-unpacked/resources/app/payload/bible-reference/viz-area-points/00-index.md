## Code Chunk (single responsibility)

```tsx
export function areaPoints(values: number[], y: (v: number) => number, baseline: number) {
  const top: Array<[number, number]> = [];
  const bottom: Array<[number, number]> = [];
  const n = Math.max(1, values.length - 1);
  values.forEach((v, i) => {
    const x = (i / n) * 100;
    top.push([x, y(v)]);
    bottom.unshift([x, baseline]);
  });
  return [...top, ...bottom];
}
```
