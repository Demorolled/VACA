## Code Chunk (single responsibility)

```tsx
export function withAudit<T extends Record<string, unknown>>(row: T, opts: AuditContext = {}) {
  const now = opts.now ?? new Date();
  const actor = opts.actor ?? null;
  const isNew = row.created_at === undefined;
  return {
    ...row,
    ...(isNew ? { created_at: now, created_by: actor } : {}),
    updated_at: now,
    updated_by: actor,
  };
}
```
