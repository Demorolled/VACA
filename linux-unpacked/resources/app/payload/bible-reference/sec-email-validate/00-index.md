```tsx
export function isValidEmail(email: string) {
  if (email.length > 254) return false;
  const m = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.exec(email.trim());
  if (!m) return false;
  const [local, domain] = email.trim().split('@');
  return local.length <= 64 && /^[a-z0-9.-]+\.[a-z]{2,}$/i.test(domain);
}
```
