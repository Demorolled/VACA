## Code Chunk (single responsibility)

```ts
export function barLayout(n: number, width: number, gapRatio = 0.2) {
  const slot = width / n;
  const w = slot * (1 - gapRatio);
  return Array.from({ length: n }, (_, i) => ({ x: i * slot + (slot - w) / 2, w }));
}
```
