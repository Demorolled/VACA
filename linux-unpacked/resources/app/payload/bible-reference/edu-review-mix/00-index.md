export function mixReview<T>(due: T[], fresh: T[], dailyNewCap: number) {
  const out: T[] = [];
  const d = due.slice(), f = fresh.slice(0, dailyNewCap);
  while (d.length || f.length) {
    if (d.length) out.push(d.shift()!);
    if (f.length) out.push(f.shift()!);
  }
  return out;
}
