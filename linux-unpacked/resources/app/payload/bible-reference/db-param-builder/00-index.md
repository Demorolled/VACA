```ts
export function paramSql(parts: string[], values: unknown[]) {
  const params: unknown[] = [];
  const sql = parts.map((p, i) => {
    if (i === parts.length - 1) return p;
    params.push(values[i]);
    return p + `$${params.length}`;
  }).join('');
  return { sql, params };
}
```
