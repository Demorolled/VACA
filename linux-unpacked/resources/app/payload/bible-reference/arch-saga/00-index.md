export async function runSaga<T>(steps: Array<SagaStep<T>>) {
  const done: Array<SagaStep<T>> = [];
  try {
    for (const s of steps) { await s.run(); done.push(s); }
    return done;
  } catch (err) {
    for (const s of done.reverse()) { try { await s.compensate(); } catch { /* best-effort */ } }
    throw err;
  }
}
