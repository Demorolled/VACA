export class Factory<K extends string, T> {
  private builders = new Map<K, () => T>();
  register(key: K, build: () => T) { this.builders.set(key, build); }
  create(key: K) {
    const b = this.builders.get(key);
    if (!b) throw new Error(`Unknown factory key: ${key}`);
    return b();
  }
}
