export function stratifiedSample<T>(items: T[], label: (t: T) => string, ratio: number, seed = 42) {
  const rand = () => {
    seed |= 0; seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
  const groups = new Map<string, T[]>();
  for (const it of items) {
    const k = label(it);
    if (!groups.has(k)) groups.set(k, []);
    groups.get(k)!.push(it);
  }
  const out: T[] = [];
  for (const g of groups.values()) {
    for (let i = g.length - 1; i > 0; i--) { const j = Math.floor(rand() * (i + 1)); [g[i], g[j]] = [g[j], g[i]]; }
    out.push(...g.slice(0, Math.max(1, Math.round(g.length * ratio))));
  }
  return out;
}
