```tsx
export function vwap(bars: Array<{ high: number; low: number; close: number; volume: number }>) {
  let pv = 0, vol = 0;
  for (const b of bars) {
    const tp = (b.high + b.low + b.close) / 3;
    pv += tp * b.volume;
    vol += b.volume;
  }
  return vol === 0 ? 0 : pv / vol;
}
```
