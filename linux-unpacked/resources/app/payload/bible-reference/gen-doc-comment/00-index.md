## Code Chunk (single responsibility)

```tsx
export function jsdoc(summary: string, params: Array<{ name: string; desc: string }> = [], returns?: string) {
  const lines = ['/**', ` * ${summary}`];
  for (const p of params) lines.push(` * @param ${p.name} ${p.desc}`);
  if (returns) lines.push(` * @returns ${returns}`);
  lines.push(' */');
  return lines.join('\n');
}
```
