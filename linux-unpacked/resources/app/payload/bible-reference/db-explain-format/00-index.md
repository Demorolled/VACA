```ts
export function formatExplain(rows: ExplainRow[]) {
  return rows.map((r) => {
    const line = r['QUERY PLAN'];
    const indent = line.match(/^\s*/)?.[0].length ?? 0;
    const cost = line.match(/cost=([\d.]+\.\.[\d.]+)/)?.[1];
    return '  '.repeat(indent) + line.trim() + (cost ? `  [cost=${cost}]` : '');
  }).join('\n');
}
```
