export async function withTransaction<T>(adapter: TransactionAdapter, work: () => Promise<T>) {
  await adapter.begin();
  try {
    const result = await work();
    await adapter.commit();
    return result;
  } catch (e) {
    try { await adapter.rollback(); } catch { /* rollback failed too */ }
    throw e;
  }
}
