  let ret: R | undefined;
  const calls: Args[] = [];
  const results: R[] = [];
  const fn = ((...args: Args) => {
    calls.push(args);
    const v = ret !== undefined ? ret : impl?.(...args);
    results.push(v as R);
    return v as R;
  }) as MockFn<Args, R>;
  fn.calls = calls;
  fn.results = results;
  fn.clear = () => { calls.length = 0; results.length = 0; };
  fn.mockReturnValue = (value: R) => { ret = value; };
  return fn;
}
