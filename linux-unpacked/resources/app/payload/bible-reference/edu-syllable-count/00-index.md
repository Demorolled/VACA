```ts
export function countSyllables(word: string) {
  const w = word.toLowerCase().replace(/[^a-z]/g, '');
  if (!w) return 0;
  let count = (w.match(/[aeiouy]+/g) ?? []).length;
  if (w.endsWith('e') && !/le$/.test(w) && count > 1) count--;
  return Math.max(1, count);
}
```
