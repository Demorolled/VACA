## Code Chunk (single responsibility)

```ts
export function recentFiles(files: FileMeta[], limit?: number) {
  const sorted = files.slice().sort((a, b) => b.modifiedMs - a.modifiedMs);
  return limit === undefined ? sorted : sorted.slice(0, limit);
}
```
