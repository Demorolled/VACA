export class StagingDir {
  private staged: string[] = [];
  constructor(private base: string) {}
  stage(name: string) {
    const path = `${this.base}/${name}`;
    this.staged.push(path);
    return path;
  }
  commit() { const out = this.staged.splice(0); return out; }
  rollback() { this.staged = []; }
}
