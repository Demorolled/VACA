```ts
export function bubbleRadius(value: number, min: number, max: number, rMin: number, rMax: number) {
  const t = max === min ? 0.5 : (value - min) / (max - min);
  return rMin + Math.sqrt(Math.max(0, Math.min(1, t))) * (rMax - rMin);
}
```
