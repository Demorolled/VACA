export async function withTransaction<T>(
  begin: () => Promise<TxExecutor>,
  commit: () => Promise<void>,
  rollback: () => Promise<void>,
  fn: (tx: TxExecutor) => Promise<T>,
): Promise<T> {
  const tx = await begin();
  try {
    const result = await fn(tx);
    await commit();
    return result;
  } catch (e) {
    try { await rollback(); } catch { /* best effort */ }
    throw e;
  }
}
