```ts
export function curveScores(scores: number[], targetMean: number) {
  const mean = scores.reduce((s, x) => s + x, 0) / (scores.length || 1);
  const shift = targetMean - mean;
  return scores.map((s) => Math.min(100, Math.max(0, s + shift)));
}
```
