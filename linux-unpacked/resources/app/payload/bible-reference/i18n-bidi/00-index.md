```ts
const RTL_RE = /[\u0590-\u08FF\uFB1D-\uFDFD\uFE70-\uFEFC]/;
export function bidiGrouped(text: string) {
  const out: string[] = [];
  let buf = '', dir: 'rtl' | 'ltr' | '' = '';
  for (const ch of text) {
    const d = RTL_RE.test(ch) ? 'rtl' : /[a-zA-Z0-9]/.test(ch) ? 'ltr' : '';
    if (d && d !== dir) { if (buf) out.push(buf); buf = ''; dir = d; }
    buf += ch;
  }
  if (buf) out.push(buf);
  return out;
}
```
