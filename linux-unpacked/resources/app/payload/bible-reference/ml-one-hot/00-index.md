```ts
export class OneHotEncoder {
  private cats: string[] = [];
  fit(values: string[]) { this.cats = [...new Set(values)].sort(); }
  transform(values: string[]) {
    return values.map((v) => {
      const row = new Array(this.cats.length).fill(0);
      const i = this.cats.indexOf(v);
      if (i >= 0) row[i] = 1;
      return row;
    });
  }
  get categories() { return this.cats; }
}
```
