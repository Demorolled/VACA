```ts
export class MovingAverage {
  private buf: number[] = [];
  private sum = 0;
  constructor(private window: number) {}
  push(sample: number) {
    this.buf.push(sample);
    this.sum += sample;
    if (this.buf.length > this.window) this.sum -= this.buf.shift()!;
    return this.sum / this.buf.length;
  }
}
```
