export function encodeLeb128(value: number) {
  const out: number[] = [];
  let v = value >>> 0;
  do {
    let b = v & 0x7f;
    v >>>= 7;
    if (v) b |= 0x80;
    out.push(b);
  } while (v);
  return out;
}
export function decodeLeb128(bytes: number[] | Uint8Array, offset = 0) {
  let value = 0, shift = 0, i = offset;
  while (i < bytes.length) {
    const b = bytes[i++];
    value |= (b & 0x7f) << shift;
    if (!(b & 0x80)) break;
    shift += 7;
    if (shift > 28) break;
  }
  return { value: value >>> 0, next: i };
}
